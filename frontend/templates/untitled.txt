<script>
    // Show the selected filename before uploading.
    const fileInput = document.getElementById("file");
    const selectedFile = document.getElementById("selected-file");

    if (fileInput && selectedFile) {
        fileInput.addEventListener("change", function () {
            selectedFile.textContent = this.files.length
                ? this.files[0].name
                : "No file selected";
        });
    }

    // Search the currently displayed rows.
    const searchInput = document.getElementById("table-search");
    const tableRows = document.querySelectorAll("#review-table tbody .data-row");

    if (searchInput) {
        searchInput.addEventListener("input", function () {
            const query = this.value.toLowerCase().trim();

            tableRows.forEach(function (row) {
                row.style.display = row.textContent.toLowerCase().includes(query)
                    ? ""
                    : "none";
            });
        });
    }

    // Update the status select's appearance when its value changes.
    document.querySelectorAll(".status-select").forEach(function (select) {
        function updateStatusStyle() {
            select.classList.remove("status-pending", "status-incorrect", "status-valid");

            if (select.value === "Confirmed incorrect") {
                select.classList.add("status-incorrect");
            } else if (select.value === "Valid label") {
                select.classList.add("status-valid");
            } else {
                select.classList.add("status-pending");
            }
        }

        select.addEventListener("change", updateStatusStyle);
        updateStatusStyle();
    });
</script>
